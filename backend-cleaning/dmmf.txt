  - userId: String (req)
  - user: User (req) [rel:LedgerEntryToUser]
  - bookingId: String (opt)
  - booking: Booking (opt) [rel:BookingToLedgerEntry]
  - disputeId: String (opt)
  - dispute: Dispute (opt) [rel:DisputeToLedgerEntry]
  - amount: Decimal (req)
  - type: LedgerEntryType (req)
  - note: String (opt)
  - createdAt: DateTime (req)
MODEL Payout
  - id: String (req) id
  - userId: String (req)
  - user: User (req) [rel:PayoutToUser]
  - amount: Decimal (req)
  - status: PayoutStatus (req)
  - gatewayTxnId: String (opt) unique
  - idempotencyKey: String (req) unique
  - requestedAt: DateTime (req)
  - processedAt: DateTime (opt)
MODEL WebhookReplay
  - id: String (req) id
  - source: String (req)
  - eventId: String (req) unique
  - seenAt: DateTime (req)
